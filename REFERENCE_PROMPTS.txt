Step 2: In backend/app/main.py, add this middleware BEFORE the rate limiter

  @app.middleware("http")
  async def csrf_protection(request: Request, call_next):
      SAFE_METHODS = {"GET", "HEAD", "OPTIONS"}
      if request.method not in SAFE_METHODS:
          origin = request.headers.get("origin", "")
          referer = request.headers.get("referer", "")
          source = origin or referer
          if source and not any(source.startswith(o) for o in settings.ALLOWED_ORIGINS):
              return JSONResponse(
                  status_code=403,
                  content={"error": {"code": "CSRF_REJECTED",
                                     "message": "Cross-site request rejected."}}
              )
      return await call_next(request)

  import hashlib, secrets as _secrets

  def create_refresh_token() -> tuple[str, str]:
      """Returns (raw_token, hashed_token). Store hash, send raw to client."""
      raw = _secrets.token_hex(64)
      hashed = hashlib.sha256(raw.encode()).hexdigest()
      return raw, hashed

  def verify_refresh_token(raw: str, hashed: str) -> bool:
      return hashlib.sha256(raw.encode()).hexdigest() == hashed

Step 6: Add POST /auth/refresh endpoint:
  @app.post("/auth/refresh")
  async def refresh_access_token(
      refresh_token: str = Cookie(default=None),
      db: Session = Depends(get_db)
  ):
      if not refresh_token:
          raise HTTPException(401, detail={"code":"TOKEN_MISSING","message":"No refresh token."})
      # Find AuthToken records where refresh_expires_at > now
      record = db.query(AuthToken).filter(
          AuthToken.refresh_expires_at > datetime.now(timezone.utc)
      ).all()
      # Find the record whose hashed refresh_token matches
      matched = next((r for r in record if verify_refresh_token(refresh_token, r.refresh_token or "")), None)
      if not matched:
          raise HTTPException(401, detail={"code":"TOKEN_INVALID","message":"Refresh token invalid or expired."})
      # Rotate: generate new pair
      raw_refresh, hashed_refresh = create_refresh_token()
      matched.refresh_token = hashed_refresh
      matched.refresh_expires_at = datetime.now(timezone.utc) + timedelta(days=settings.REFRESH_TOKEN_EXPIRE_DAYS)
      db.commit()
      # Issue new access token
      new_access = create_token({"sub": str(matched.user_id)})
      response = JSONResponse({"message": "Token refreshed"})
      response.set_cookie("access_token", new_access, httponly=True, secure=True, samesite="strict",
                          max_age=settings.ACCESS_TOKEN_EXPIRE_MINUTES * 60)
      response.set_cookie("refresh_token", raw_refresh, httponly=True, secure=True, samesite="strict",
                          max_age=settings.REFRESH_TOKEN_EXPIRE_DAYS * 86400, path="/api/auth")
      return response

      question_id: int
      selected: Optional[str] = None  # None = skipped

  class MockTestSubmission(BaseModel):
      session_id: str
      answers: list[SubmitAnswer]

  @app.post("/mock-tests/{test_id}/submit")
  @limiter.limit("10/minute")
  async def submit_mock_test(
      request: Request,
      test_id: int,
      body: MockTestSubmission,
      current_user: User = Depends(get_current_user),
      db: Session = Depends(get_db)
  ):

      session = db.query(MockTestSession).filter(
          MockTestSession.id == body.session_id,
          MockTestSession.user_id == current_user.id
      ).first()
      if not session:
          raise api_error(404, "SESSION_NOT_FOUND", "Session not found.")
      if session.expires_at < datetime.now(timezone.utc):
          raise api_error(400, "SESSION_EXPIRED", "Test session has expired.")

      questions = db.query(MasterQuestion).filter(MasterQuestion.test_id == test_id).all()

      CORRECT_MARKS = 1.0
      NEGATIVE_MARKS = 0.25
      score = 0.0
      results = []

      for ans in body.answers:

              mark = 0.0  # skipped

              is_correct = True
          else:
              mark = -NEGATIVE_MARKS
          score += mark
          results.append(format_question_payload(q, mode="result") | {
              "user_selected": ans.selected,
              "is_correct": is_correct,
              "marks_awarded": mark,
          })

      final_score = max(0.0, score)
      total = len(questions)
      percentage = round((final_score / total) * 100, 1) if total > 0 else 0.0

      # 4. Save result
      save_mock_test(db, current_user.id, test_id, int(final_score), total, results)

      # 5. Invalidate session
      db.delete(session)
      db.commit()

      return {
          "score": final_score,
          "total": total,
          "percentage": percentage,
          "negative_marks_applied": True,
          "results": results,
      }
