Files: backend/app/main.py, backend/app/core/config.py, backend/.env.example

1. Add CSRF middleware (before rate limiter):
   @app.middleware("http")
   async def csrf_protection(request: Request, call_next):
       SAFE_METHODS = {"GET", "HEAD", "OPTIONS"}
       if request.method not in SAFE_METHODS:
           origin = request.headers.get("origin", "")
           referer = request.headers.get("referer", "")
           allowed_origins = settings.ALLOWED_ORIGINS
           source = origin or referer
           if not any(source.startswith(o) for o in allowed_origins):
               return JSONResponse(status_code=403,
                   content={"error":{"code":"CSRF_REJECTED","message":"Cross-site request rejected."}})
       return await call_next(request)
